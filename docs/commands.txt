# Commands used for electron-builder audit
# Run from git bash (mingw64) on Windows

# 1. List generated POS directories
ls -lt "D:/Carthapos/backend/generated-pos/" | head -10

# 2. Find which generated dirs have release directories (full builds)
for d in "D:/Carthapos/backend/generated-pos/"*/; do
  r=$(ls "$d/release/"*.exe 2>/dev/null | head -1)

done
